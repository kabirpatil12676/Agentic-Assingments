Task 19: ReAct + Tool + Reasoning
Objective:
Create a simple ReAct-style agent that decides whether to use a multiplication tool, executes the tool, observes the result, and generates a final answer.
Code:
import os
from dotenv import load_dotenv
from huggingface_hub import InferenceClient from langchain_core.tools import tool load_dotenv()
token = os.getenv("HF_TOKEN")

# Create Hugging Face client client = InferenceClient(
api_key=token, provider="featherless-ai"
) #
# Tool #
@tool
def multiply(a: int, b: int): """Multiply two numbers.""" return a * b
#
# Agent #
user_task = input("Enter your request: ") agent_prompt = f"""
You are an AI agent with access to one tool: Tool:
multiply(a, b) - multiplies two numbers. User request:
{user_task}
Decide whether the multiply tool is needed.
If multiplication is needed, respond EXACTLY in this format:
ACTION: multiply a: number
b: number
If the tool is not needed, respond EXACTLY: ACTION: none
Do not provide the final answer yet. """
# Agent decides what to do
response = client.chat.completions.create( model="meta-llama/Llama-3.2-1B-Instruct", messages=[
{
"role": "user", "content": agent_prompt
}

],
max_tokens=80
)
decision = response.choices[0].message.content.strip() print("\nAGENT DECISION:")
print(decision) #
# Tool execution #
if "ACTION: multiply" in decision:


try:

lines = decision.splitlines()

a = int(lines[1].split(":")[1].strip())
b = int(lines[2].split(":")[1].strip()) print("\nACTION: multiply")
result = multiply.invoke({ "a": a,
"b": b

})
print("OBSERVATION:", result) #
# Agent gives final answer #

final_prompt = f""" User request:
{user_task}
You decided to use the multiply tool. Tool result:
{result}
Give the final answer to the user. Keep it short.
"""
final_response = client.chat.completions.create( model="meta-llama/Llama-3.2-1B-Instruct", messages=[
{
"role": "user", "content": final_prompt
}
],
max_tokens=80
)

final_answer = final_response.choices[0].message.content print("\nFINAL ANSWER:")

print(final_answer)

except Exception as e:
print("\nTool execution failed.") print("Error:", e)
else:
print("\nFINAL ANSWER:")
print("No multiplication tool was required.")
