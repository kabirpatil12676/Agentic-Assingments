The program loads the Hugging Face token from a .env file, sends formatted prompts to the Featherless AI 
provider, and uses the Llama 3.2 1B Instruct model for all four tasks.  
import os 
from dotenv import load_dotenv  
  
from huggingface_hub import InferenceClient  
from langchain_core.prompts import PromptTemplate  
  
# Load environment variables  
load_dotenv() 
  
token = os.getenv("HF_TOKEN")  
  
print("Token loaded:", bool(token))  
  
# Hugging Face client  
client = InferenceClient(  
    api_key=token, 
    provider="featherless -ai" 
) 
  
# Model 
model = "meta-llama/Llama-3.2-1B-Instruct" 
  
  
def generate_response(prompt, max_tokens):  
    response = client.chat.completions.create(  
        model=model, 
        messages=[ 
            { 
                "role": "user", 
                "content": prompt  
            } 
        ], 
        max_tokens=max_tokens  
    ) 
  
    return response.choices[0].message.content

    print("\n==============================")

    # -----------------------------------
