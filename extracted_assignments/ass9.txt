from huggingface_hub import InferenceClient

provider="featherless-ai" 
)

Task 22: API Weather Integration and LLM Advice Agent 
Objective: 
Create an agent that fetches real-time weather data for a city using an external API and uses an LLM to 
generate concise, practical user advice. 
Code: 
import os 
import requests 
from dotenv import load_dotenv 
from huggingface_hub import InferenceClient 
 
load_dotenv() 
token = os.getenv("HF_TOKEN") 
 
client = InferenceClient( 
api_key=token, 
provider="featherless-ai" 
) 
city = input("Enter city: ") 
# Get weather data from API 
url = f"https://wttr.in/{city}?format=j1" 
data = requests.get(url).json() 
 
temp = data["current_condition"][0]["temp_C"] 
weather = data["current_condition"][0]["weatherDesc"][0]["value"] 
 
# LLM Agent 
prompt = f""" 
You are a weather advice agent. 
Give simple advice for someone in {city}. 
Weather: {weather} 
Temperature: {temp} C 
 
Keep the advice short and practical. 
""" 
 
response = client.chat.completions.create( 
model="meta-llama/Llama-3.2-1B-Instruct", 
messages=[ 
{"role": "user", "content": prompt} 
], 
max_tokens=100 
) 
 
print("\nWeather:") 
print(weather)

print("Temperature:", temp, "C") 
 
print("\nAgent Advice:") 
print(response.choices[0].message.content)

Task 23: Computer Vision Classification and Decision Threshold Agent 
Objective: 
Create an agent that classifies images using a pre-trained Vision Transformer model and makes an 
automated decision based on confidence thresholding. 
Code: 
from pathlib import Path 
from PIL import Image 
from transformers import pipeline 
 
classifier = pipeline( 
"image-classification", 
model="google/vit-base-patch16-224" 
) 
image_path = Path(input("Enter image path: ").strip().strip('"').strip("'")) 
if not image_path.is_file(): 
raise FileNotFoundError(f"Image not found: {image_path.resolve()}") 
 
with Image.open(image_path) as image: 
results = classifier(image.convert("RGB")) 
 
label = results[0]["label"] 
score = results[0]["score"] 
 
print("\nPrediction:", label) 
print("Confidence:", round(score, 2)) 
 
if score > 0.7: 
print("Agent Action: Accept prediction") 
else: 
print("Agent Action: Request human review")
