Task 6: Calculator Agent
Objective: Create an agent that can use a calculator tool..
Code

Task 10: Temperature Comparison
Objective: Generate responses using low and high temperature settings and compare how temperature affects creativity and variation.
Code
import osfrom dotenv import load_dotenvfrom huggingface_hub import InferenceClientload_dotenv()token = os.getenv("HF_TOKEN")client = InferenceClient(    api_key=token,    provider="featherless-ai")model = "meta-llama/Llama-3.2-1B-Instruct"prompt = "Write a creative sentence about Artificial Intelligence."low = client.chat.completions.create(    model=model,    messages=[        {"role": "user", "content": prompt}    ],    temperature=0.1,    max_tokens=50)high = client.chat.completions.create(    model=model,    messages=[        {"role": "user", "content": prompt}    ],    temperature=1.0,    max_tokens=50)print("Low temperature:")print(low.choices[0].message.content)print("\nHigh temperature:")print(high.choices[0].message.content)
